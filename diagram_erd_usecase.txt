2. ENTITY RELATIONSHIP DIAGRAM (ERD)

erDiagram
    %% RELATIONSHIPS
    CATEGORIES ||--o{ MENUS : "memiliki (has)"
    ROOMS ||--o{ ORDERS : "melayani (serves)"
    ORDERS ||--|{ ORDER_ITEMS : "berisi (contains)"
    MENUS ||--o{ ORDER_ITEMS : "termasuk di (included in)"
    ORDERS ||--o| PAYMENTS : "memiliki (has)"
    USERS ||--o{ ORDERS : "membuat (opsional)"

    %% TABLES & COLUMNS
    USERS {
        bigint id PK
        string name
        string email
        string password
        string role "enum: owner, kitchen, finance"
        datetime created_at
    }
    
    CATEGORIES {
        bigint id PK
        string name
        string description
        datetime created_at
    }
    
    MENUS {
        bigint id PK
        bigint category_id FK
        string name
        string slug
        decimal price
        string image_path
        boolean is_available
        datetime created_at
    }
    
    ROOMS {
        bigint id PK
        string room_number
        integer capacity
        string qr_code
        datetime created_at
    }
    
    ORDERS {
        bigint id PK
        bigint room_id FK
        bigint user_id FK "Nullable"
        string status "pending, cooking, ready, delivered, completed"
        decimal total_amount
        datetime created_at
    }
    
    ORDER_ITEMS {
        bigint id PK
        bigint order_id FK
        bigint menu_id FK
        integer quantity
        decimal price
        decimal subtotal
        string notes
    }
    
    PAYMENTS {
        bigint id PK
        bigint order_id FK
        string payment_method
        string payment_status "pending, success, failed"
        decimal amount
        datetime paid_at
    }
